85:    @classmethod
86:    def from_line(cls, line: str) -> Note | None:
87:        """从文本行解析 Note，解析失败返回 None"""
88:        line = line.strip()
89:        if not line or line.startswith("#"):
90:            return None
91:        # 跳过注释行（只有 # 开头且第二位不是数字的才是注释）
92:        if len(line) > 1 and line[0] == "#" and not line[1].isdigit():
93:            return None
94:        try:
95:            # 格式: [#id ]轨道名  时间	| 内容
96:            rest = line
97:            # 跳过可选的 #id 前缀
98:            if rest[0] == "#":
99:                space_pos = rest.find(" ")
100:                if space_pos > 1 and rest[1:space_pos].isdigit():
101:                    rest = rest[space_pos+1:]
102:            
103:            parts = rest.split("	| ", 1)
104:            if len(parts) != 2:
105:                return None
106:            meta, text = parts
107:            meta_parts = meta.rsplit("	", 1)
108:            if len(meta_parts) != 2:
109:                return None
110:            track_name, time_str = meta_parts
111:            if track_name not in NOTE_TYPES:
112:                return None
113:            m, r = time_str.split(":", 1)
114:            s, cs = r.split(".")
115:            ms = int(m) * 60000 + int(s) * 1000 + int(cs) * 10
116:            return cls(timestamp_ms=ms, text=text, type=track_name)
117:        except Exception:
118:            return None

120:@dataclass
121:class Term:
122:    """术语条目"""
123:    source: str      # 原文（日语）
124:    translation: str # 译文（中文）
125:    origin: str = "" # 出处
126:    note: str = ""   # 备注
127:
128:    def to_line(self) -> str:
129:        """转导出文本行"""
130:        parts = [self.source, self.translation]
131:        if self.origin:
132:            parts.append(self.origin)
133:        if self.note:
134:            parts.append(self.note)
135:        return " | ".join(parts)

137:    @classmethod
138:    def from_line(cls, line: str) -> Term | None:
139:        """从文本行解析 Term"""
140:        line = line.strip()
141:        if not line or line.startswith("#"):
142:            return None
143:        try:
144:            parts = [p.strip() for p in line.split(" | ", 3)]
145:            if len(parts) < 2:
146:                return None
147:            term = cls(source=parts[0], translation=parts[1])
148:            if len(parts) > 2:
149:                term.origin = parts[2]
150:            if len(parts) > 3:
151:                term.note = parts[3]
152:            return term
153:        except Exception:
154:            return None
